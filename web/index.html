Step A: Setup PostgreSQL Schema

Connect to your own Supabase project dashboard, open the **SQL Editor**, create a new query page, paste the following schema layout, and click **Run**. 

-- Create SwapnoPay Postgres tables
CREATE TABLE IF NOT EXISTS orders (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  amount NUMERIC NOT NULL,
  status VARCHAR(20) DEFAULT 'PENDING', -- PENDING, PAID, CANCELLED
  tran_id VARCHAR(50) UNIQUE,
  cus_phone VARCHAR(20) NOT NULL,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

CREATE TABLE IF NOT EXISTS appeals (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  order_id UUID REFERENCES orders(id),
  trx_id VARCHAR(50) NOT NULL,
  cus_phone VARCHAR(20) NOT NULL,
  note TEXT,
  status VARCHAR(30) DEFAULT 'PENDING_REVIEW'
);

Step D: Add Live Database Triggers

Hook raw SMS logs insert operations to execute processing tasks automatically. Paste this final trigger inside your Supabase SQL window. 

-- Create matching triggers
CREATE OR REPLACE TRIGGER on_sms_received
  AFTER INSERT ON sms_logs
  FOR EACH ROW
  EXECUTE FUNCTION invoke_process_sms_gateway();
